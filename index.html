// 1. Install via npm
npm install lanka-kit

// 2. Validate and extract NIC details
import { validateNic, validatePhone, lookupBank, formatLkr, amountToWords } from 'lanka-kit';

const identity = validateNic('931451234V');
if (identity.isValid) {
  console.log(identity.gender);               // "Male"
  console.log(identity.dateOfBirthFormatted); // "1993-05-24"
  console.log(identity.age);                  // 33
  console.log(identity.equivalentNic);        // "199314501234"
}

// 3. Banking & Currency
const bank = lookupBank('7010');            // Commercial Bank
const lkr = formatLkr(1540.50);              // "Rs. 1,540.50"
const words = amountToWords(1540.50);        // "One Thousand Five Hundred Forty Rupees and Fifty Cents Only"

// 4. Zod Schema Integration
import { z } from 'zod';
import { nicRefinement, phoneRefinement } from 'lanka-kit';

const RegisterSchema = z.object({
  nic: z.string().refine(nicRefinement.validator, { message: nicRefinement.message }),
  phone: z.string().refine(phoneRefinement.validator, { message: phoneRefinement.message }),
});
// 1. Add to Flutter project:
flutter pub add lanka_kit

// 2. Import in Flutter Widgets:
import 'package:lanka_kit/lanka_kit.dart';

// Use directly in TextFormField validator
TextFormField(
  decoration: InputDecoration(labelText: 'Sri Lankan NIC'),
  validator: nicFormFieldValidator(),
);

// Or parse programmatically
final res = validateNic('931451234V');
if (res.isValid) {
  print('Gender: ${res.gender}');           // Male
  print('DOB: ${res.dateOfBirth}');          // 1993-05-24
  print('New NIC: ${res.equivalentNic}');   // 199314501234
}

// Phone Carrier & E.164 Normalization
final phone = validatePhone('0771234567');
print(phone.carrier);                        // "Dialog"
print(phone.e164);                           // "+94771234567"

// 2. Java Swing Auto-Fill & Validation Example
JTextField nicField = new JTextField();
JTextField dobField = new JTextField();

nicField.getDocument().addDocumentListener(new DocumentListener() {
    public void update() {
        LankaKit.NicResult res = LankaKit.validateNic(nicField.getText());
        if (res.isValid) {
            dobField.setText(res.dateOfBirth); // Auto-fills DOB e.g. "1993-05-24"
        }
    }
    public void insertUpdate(DocumentEvent e) { update(); }
    public void removeUpdate(DocumentEvent e) { update(); }
    public void changedUpdate(DocumentEvent e) { update(); }
});
